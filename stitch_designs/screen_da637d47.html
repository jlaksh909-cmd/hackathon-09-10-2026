Lemma 4.3 (Strict Store Buffering Invariance) TSO Memory Order

02  _sync_epoch_latch:
03    movq    (%rdi), %rax          # Load active reservation ticket
04    lock cmpxchgq %rsi, (%rdi)  # Atomic compare & swap epoch pointer
05    jnz     .Lretry_contention    # Spin-wait on memory bus contention
06    sfence                        # Flush stores to L2 cache controller
07    retq
08  .Lretry_contention:
09    pause                         # Alleviate pipeline stall penalty
10    jmp     _sync_epoch_latch
